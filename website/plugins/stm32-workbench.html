    HAL_Init();
    SystemClock_Config();
    LED_GPIO_Config();
    LCD_Init();

    // 独家延时防锁死：无论 -O0 还是 -O3，精准保持时钟节奏，杜绝假死

        LED1_TOGGLE();
        HAL_Delay(500);   // Breakpoint #1 Hit (PC = 0x080002AC)
        LCD_DispString_EN_CH(10, 20, (uint8_t*)"SmileCode IDE Running");
    }
}
